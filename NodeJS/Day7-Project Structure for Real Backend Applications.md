## 1. Why Project Structure Matters

When you are learning, putting everything in one file works fine. But in a real project, that one file quickly grows to hundreds of lines. Finding a bug, adding a feature, or working with another developer becomes painful.

Good project structure means:

- Every file has one clear job
- You know exactly where to look when something needs to change
- A new developer can understand the project in minutes
- Testing and debugging are much easier

Think of a well-organized kitchen. Every tool has a drawer. Every ingredient has a shelf. You can cook efficiently because you know where everything is. A messy kitchen works for one quick meal but not for a professional restaurant running all day.

## 2. The Official Course Folder Structure

This is the structure every project in this course follows. Learn it well.

```
project-name/
  src/
    server.js            <- entry point: starts Express, imports routes
    controllers/
      userController.js  <- logic for user routes
      productController.js
    middlewares/
      logger.js          <- logging middleware
      errorHandler.js    <- global error handler
      auth.js            <- JWT authentication middleware
    routes/
      userRoutes.js      <- maps URLs to controller functions
      productRoutes.js
    db/
      prisma.js          <- creates and exports the Prisma client
    generated/
      prisma/            <- auto-generated by Prisma (never edit)
  prisma/
    schema.prisma        <- define your database models here
    migrations/          <- auto-created by prisma migrate
    prisma.config.js     <- Prisma config file
  .env                   <- secrets and config, never commit to Git
  .gitignore             <- files and folders Git should ignore
  docker-compose.yaml    <- PostgreSQL container definition
  package.json
```

Each folder has a single responsibility:

| Folder / File          | Responsibility                                  |
| ---------------------- | ----------------------------------------------- |
| `src/server.js`        | Starts Express, registers middleware and routes |
| `src/controllers/`     | Handles request logic, talks to database        |
| `src/middlewares/`     | Functions that run before route handlers        |
| `src/routes/`          | Maps URLs to the right controller function      |
| `src/db/prisma.js`     | Creates the Prisma database client              |
| `prisma/schema.prisma` | Defines database models and structure           |
| `docker-compose.yaml`  | Defines the PostgreSQL container                |
| `.env`                 | Stores secrets and configuration                |
| `.gitignore`           | Tells Git what not to track                     |

## 5. The Controller File

The controller contains all the logic. It reads `req`, does work, and sends `res`. It does not care about URLs - that is the router's job.

Create `src/controllers/userController.js`:

```javascript
// Import the in-memory data store
import { users, nextId, incrementId } from '../data/users.js';

// GET /users - return all users
export function getAllUsers(req, res) {
  res.status(200).json({ count: users.length, users });
}

// GET /users/:id - return one user
export function getUserById(req, res) {
  const id = parseInt(req.params.id);  // convert URL string to number
  const user = users.find(u => u.id === id);

  if (!user) {
    return res.status(404).json({ error: `User ${id} not found` });
  }

  res.status(200).json(user);
}

// POST /users - create a new user
export function createUser(req, res) {
  const { name, email, age } = req.body;  // read fields from request body

  if (!name || !email) {
    return res.status(400).json({ error: 'Name and email are required' });
  }

  // Build new user with the current nextId
  const newUser = { id: nextId, name, email, age: age || null };
  incrementId();        // update the shared counter
  users.push(newUser);  // add to the array

  res.status(201).json({ message: 'User created', user: newUser });
}

// PUT /users/:id - replace all fields of a user
export function updateUser(req, res) {
  const id = parseInt(req.params.id);
  const index = users.findIndex(u => u.id === id);  // -1 if not found

  if (index === -1) {
    return res.status(404).json({ error: `User ${id} not found` });
  }

  const { name, email, age } = req.body;

  if (!name || !email) {
    return res.status(400).json({ error: 'Name and email are required' });
  }

  // Replace entire user object at this index, keeping the same id
  users[index] = { id, name, email, age: age || null };

  res.status(200).json({ message: 'User updated', user: users[index] });
}

// DELETE /users/:id - remove a user
export function deleteUser(req, res) {
  const id = parseInt(req.params.id);
  const index = users.findIndex(u => u.id === id);

  if (index === -1) {
    return res.status(404).json({ error: `User ${id} not found` });
  }

  users.splice(index, 1);  // remove 1 element at this position

  res.status(200).json({ message: `User ${id} deleted` });
}
```

## 6. The Route File

The route file only maps URLs to controller functions. No logic here at all.

Create `src/routes/userRoutes.js`:

```javascript
import express from 'express';
import {
  getAllUsers,
  getUserById,
  createUser,
  updateUser,
  deleteUser
} from '../controllers/userController.js';

// Create a mini Express application (router)
const router = express.Router();

// Map each HTTP method + path to a controller function
// The base path /users is added in server.js when we mount this router
router.get('/', getAllUsers);        // GET /users
router.get('/:id', getUserById);    // GET /users/:id
router.post('/', createUser);       // POST /users
router.put('/:id', updateUser);     // PUT /users/:id
router.delete('/:id', deleteUser);  // DELETE /users/:id

export default router;
```

## 7. The Logger Middleware

Create `src/middlewares/logger.js`:

```javascript
// Logs every incoming request with timestamp, method, and URL
export function logger(req, res, next) {
  const time = new Date().toISOString();
  console.log(`[${time}] ${req.method} ${req.url}`);
  next();  // MUST call next() to pass control forward
}
```

## 8. The Entry Point: src/server.js

The server file only wires everything together and starts the server. It contains no route logic.

```javascript
// Load .env values first - always the very first import
import 'dotenv/config';
import express from 'express';

// Import routes and middleware
import userRoutes from './routes/userRoutes.js';
import { logger } from './middlewares/logger.js';

const app = express();

// Built-in middleware: parse JSON bodies
app.use(express.json());

// Custom global middleware: log every request
app.use(logger);

// Mount the user router at /users
// All routes inside userRoutes.js become prefixed with /users
// So router.get('/') becomes GET /users
// And router.get('/:id') becomes GET /users/:id
app.use('/users', userRoutes);

// 404 fallback: runs if nothing above matched the URL
app.use((req, res) => {
  res.status(404).json({ error: 'Route not found' });
});

const PORT = process.env.PORT || 8888;
app.listen(PORT, () => {
  console.log(`Server running at http://localhost:${PORT}`);
});
```

## Summary

Here is what you covered today:

- Every project uses `src/server.js` as the entry point, not `app.js`.
- Routes go in `src/routes/`, controllers go in `src/controllers/`, middleware in `src/middlewares/`, database client in `src/db/`.
- Prisma schema and migrations live in a `prisma/` folder at the root level.
- `docker-compose.yaml` and `.env` stay at the root level.
- `express.Router()` creates a modular router. Mount it in `server.js` with `app.use('/path', router)`.
- The server file only wires things together. It contains no business logic.

#### Step 2 — Create `src/controllers/productController.js`

Move product request and response logic out of the server.

**File: `src/controllers/productController.js`**

~~~javascript
import { createProductId, products } from '../data/products.js';

export function getAllProducts(req, res) {
  res.json({ data: products });
}

export function getProductById(req, res) {
  const product = products.find(item => item.id === Number(req.params.id));
  if (!product) return res.status(404).json({ message: 'Product not found' });
  res.json({ data: product });
}

export function createProduct(req, res) {
  const { title, price, description = '' } = req.body;
  if (!title || typeof price !== 'number') {
    return res.status(400).json({ message: 'title and numeric price are required' });
  }
  const product = { id: createProductId(), title, price, description };
  products.push(product);
  res.status(201).json({ data: product });
}

export function updateProduct(req, res) {
  const index = products.findIndex(item => item.id === Number(req.params.id));
  if (index === -1) return res.status(404).json({ message: 'Product not found' });
  const { title, price, description = '' } = req.body;
  if (!title || typeof price !== 'number') {
    return res.status(400).json({ message: 'title and numeric price are required' });
  }
  products[index] = { ...products[index], title, price, description };
  res.json({ data: products[index] });
}

export function deleteProduct(req, res) {
  const index = products.findIndex(item => item.id === Number(req.params.id));
  if (index === -1) return res.status(404).json({ message: 'Product not found' });
  products.splice(index, 1);
  res.status(204).send();
}
~~~

This is the complete Level 7 version of `src/controllers/productController.js`. Move product request and response logic out of the server. Save it at exactly this path before continuing; imports in the checkpoint assume this location.
